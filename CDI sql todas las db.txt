PRAGMA foreign_keys = OFF;
DROP TABLE IF EXISTS documentos_expedidos;
DROP TABLE IF EXISTS ciudadanos;

DROP TABLE IF EXISTS zonas_votacion;
DROP TABLE IF EXISTS ciudades;
DROP TABLE IF EXISTS usuario;
PRAGMA foreign_keys = ON;

CREATE TABLE ciudades (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    nombre TEXT NOT NULL,
    departamento TEXT DEFAULT 'Boyacá',
    codigo_dane TEXT UNIQUE
);

CREATE TABLE zonas_votacion (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    id_ciudad INTEGER,
    nombre_zona TEXT NOT NULL,
    puesto_votacion TEXT NOT NULL,
    direccion TEXT NOT NULL,
    FOREIGN KEY (id_ciudad) REFERENCES ciudades(id)
);

CREATE TABLE mesas_votacion (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    id_zona INTEGER,
    numero_mesa INTEGER NOT NULL,
    capacidad INTEGER DEFAULT 200,
    FOREIGN KEY (id_zona) REFERENCES zonas_votacion(id)
);

CREATE TABLE ciudadanos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    numero_documento TEXT NOT NULL UNIQUE,
    nombres TEXT NOT NULL,
    apellidos TEXT NOT NULL,
    fecha_nacimiento DATE NOT NULL,
    vereda_barrio TEXT NOT NULL,
    telefono TEXT,
    correo TEXT,
    fecha_registro DATETIME DEFAULT CURRENT_TIMESTAMP,
    id_mesa INTEGER,
    FOREIGN KEY (id_mesa) REFERENCES mesas_votacion(id)
);

CREATE TABLE documentos_expedidos (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    id_ciudadano INTEGER,
    tipo_documento TEXT NOT NULL,
    numero_serie TEXT NOT NULL UNIQUE,
    fecha_expedicion DATE NOT NULL,
    fecha_vencimiento DATE,
    estado TEXT DEFAULT 'vigente',
    observaciones TEXT,
    FOREIGN KEY (id_ciudadano) REFERENCES ciudadanos(id)
);

CREATE TABLE usuario (
    idUsuario INTEGER PRIMARY KEY AUTOINCREMENT,
    documento TEXT,
    nombre TEXT,
    apellido TEXT,
    email TEXT UNIQUE,
    telefono TEXT,
    estado INTEGER DEFAULT 1,
    password TEXT
);

CREATE TABLE ciudades (
    id INT IDENTITY(1,1) PRIMARY KEY,

    id INT IDENTITY(1,1) PRIMARY KEY,
    id_ciudad INT,
    nombre_zona NVARCHAR(100),
    puesto_votacion NVARCHAR(100),
    direccion NVARCHAR(255),

    id INT IDENTITY(1,1) PRIMARY KEY,
    id_zona INT,
    numero_mesa INT,
    capacidad INT DEFAULT 200,
    FOREIGN KEY (id_zona) REFERENCES zonas_votacion(id)

    numero_documento NVARCHAR(20) UNIQUE,
    nombres NVARCHAR(100),
    apellidos NVARCHAR(100),

    vereda_barrio NVARCHAR(255),
    telefono NVARCHAR(20),
    correo NVARCHAR(100),
    fecha_registro DATETIME DEFAULT GETDATE(),
